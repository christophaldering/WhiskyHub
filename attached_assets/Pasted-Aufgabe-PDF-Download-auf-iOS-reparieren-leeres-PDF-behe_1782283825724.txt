  if (typeof navigator === "undefined") return false;
  const ua = navigator.userAgent || "";
  return /iP(hone|ad|od)/i.test(ua) || ((navigator as any).platform === "MacIntel" && (navigator as any).maxTouchPoints > 1);
}
