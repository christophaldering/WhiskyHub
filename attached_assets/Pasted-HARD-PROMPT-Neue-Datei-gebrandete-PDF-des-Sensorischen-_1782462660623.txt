// client/src/lib/cooperMemoryPdf.ts
// Einseitige, CaskSense-gebrandete PDF des "Sensorischen Gedächtnisses" (Cooper-Porträt).
// Druck-Palette wie die Tasting-Sheets: Gold auf Weiß, Serifen-Body.
import jsPDF from "jspdf";
import { saveJsPdf } from "@/lib/pdf";

type RGB = [number, number, number];
const GOLD: RGB = [139, 105, 20];
const DARK: RGB = [26, 18, 8];
const MUTED: RGB = [136, 128, 112];

function goldLine(doc: jsPDF, x: number, y: number, width: number, opacity = 0.5) {
  const r = Math.round(139 * opacity + 255 * (1 - opacity));
  const g = Math.round(105 * opacity + 255 * (1 - opacity));
  const b = Math.round(20 * opacity + 255 * (1 - opacity));
  doc.setDrawColor(r, g, b);
  doc.setLineWidth(0.4);
  doc.line(x, y, x + width, y);
}

function fmtDate(iso: string | null): string {
  if (!iso) return "";
  try { return new Date(iso).toLocaleDateString("de-DE", { day: "numeric", month: "long", year: "numeric" }); }
  catch { return ""; }
}

export async function downloadCooperMemoryPdf(memoryText: string, updatedAt: string | null, participantName?: string): Promise<void> {
  const doc = new jsPDF({ unit: "mm", format: "a4" });
  const pageW = 210, pageH = 297, mx = 22, contentW = pageW - mx * 2;

  doc.setFont("helvetica", "bold");
  doc.setFontSize(10);
  doc.setTextColor(...GOLD);
  doc.text("CASKSENSE", mx, 24, { charSpace: 2 });

  doc.setFont("times", "bold");
  doc.setFontSize(24);
  doc.setTextColor(...DARK);
  doc.text("Sensorisches Gedächtnis", mx, 40);

  doc.setFont("helvetica", "normal");
  doc.setFontSize(9.5);
  doc.setTextColor(...MUTED);
  const sub = `${participantName ? participantName + " · " : ""}Stand ${fmtDate(updatedAt)}`.trim();
  if (sub) doc.text(sub, mx, 47);

  goldLine(doc, mx, 52, contentW, 0.5);

  doc.setFont("times", "normal");
  doc.setFontSize(12.5);
  doc.setTextColor(...DARK);
  const lines = doc.splitTextToSize(memoryText || "", contentW);
  doc.text(lines, mx, 62, { lineHeightFactor: 1.55 });

  goldLine(doc, mx, pageH - 22, contentW, 0.35);
  doc.setFont("helvetica", "normal");
  doc.setFontSize(8);
  doc.setTextColor(...MUTED);
  doc.text("casksense.com", mx, pageH - 16);
  doc.text("Nur in deinem Bereich gespeichert.", pageW - mx, pageH - 16, { align: "right" });

  await saveJsPdf(doc, "casksense-sensorisches-gedaechtnis.pdf");
}
