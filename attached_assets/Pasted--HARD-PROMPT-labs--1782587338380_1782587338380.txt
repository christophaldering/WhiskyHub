def splice(text,anchor,newblocks):
    a=text.find(anchor)
    if a==-1: raise SystemExit("ANKER FEHLT: "+anchor[:40])
    start=text.find('blocks: [',a)
    bp=text.find('[',start); depth=0; i=bp; ins=False; esc=False; q=''
    while i<len(text):
        c=text[i]
        if ins:
            if esc: esc=False
            elif c=='\\\\': esc=True
            elif c==q: ins=False
        else:
            if c in ('"',"'"): ins=True; q=c
            elif c=='[': depth+=1
            elif c==']':
                depth-=1
                if depth==0: break
        i+=1
    return text[:start]+newblocks+text[i+1:]
