def splice(text,anchor,nb):
    a=text.find(anchor)
    if a==-1: raise SystemExit("ANKER FEHLT: "+anchor[:40])
    start=text.find('blocks: [',a); bp=text.find('[',start)
    depth=0;i=bp;ins=False;esc=False;q=''
    while i<len(text):
        c=text[i]
        if ins:
            if esc:esc=False
            elif c=='\\':esc=True
            elif c==q:ins=False
        else:
            if c in ('"',"'"):ins=True;q=c
            elif c=='[':depth+=1
            elif c==']':
                depth-=1
                if depth==0:break
        i+=1
    return text[:start]+nb+text[i+1:]

def del_story(text,label):
    li=text.find(label)
    if li==-1: print("  (landing.story bereits weg:",label[:24],")"); return text
    sp=text.rfind('story: {',0,li); bp=text.find('{',sp)
    depth=0;i=bp;ins=False;esc=False;q=''
    while i<len(text):
        c=text[i]
        if ins:
            if esc:esc=False
            elif c=='\\':esc=True
            elif c==q:ins=False
        else:
            if c in ('"',"'"):ins=True;q=c
            elif c=='{':depth+=1
            elif c=='}':
                depth-=1
                if depth==0:break
        i+=1
    j=i+1
    if j<len(text) and text[j]==',': j+=1
    ls=text.rfind('\n',0,sp); k=j
    if k<len(text) and text[k]=='\n': k+=1
    return text[:ls]+text[k:]
