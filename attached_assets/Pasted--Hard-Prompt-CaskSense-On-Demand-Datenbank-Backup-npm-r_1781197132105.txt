## Schritt 1 — Backup-Skript anlegen

Neue Datei `script/db-backup.ts` mit exakt diesem Inhalt:

```ts
/**
 * CaskSense – On-Demand Datenbank-Backup
 * Erzeugt einen komprimierten pg_dump (custom format) unter backups/.
 * Aufruf:  npm run db:backup
 * Restore: pg_restore -d "$DATABASE_URL" --clean --if-exists backups/<datei>.dump
 */
import { spawn } from "node:child_process";
import { mkdirSync, readdirSync, statSync, unlinkSync } from "node:fs";
import { join } from "node:path";

const DATABASE_URL = process.env.DATABASE_URL;
if (!DATABASE_URL) {
  console.error("[db:backup] DATABASE_URL must be set");
  process.exit(1);
}

const RETENTION = 10;        // wie viele lokale Dumps behalten
const BACKUP_DIR = "backups";

function berlinStamp(): string {
  const parts = new Intl.DateTimeFormat("sv-SE", {
    timeZone: "Europe/Berlin",
    year: "numeric", month: "2-digit", day: "2-digit",
    hour: "2-digit", minute: "2-digit", hour12: false,
  }).formatToParts(new Date());
  const get = (t: string) => parts.find((p) => p.type === t)?.value ?? "00";
  return `${get("year")}-${get("month")}-${get("day")}_${get("hour")}${get("minute")}`;
}

mkdirSync(BACKUP_DIR, { recursive: true });
const outFile = join(BACKUP_DIR, `casksense-${berlinStamp()}.dump`);

console.log(`[db:backup] Starte pg_dump → ${outFile}`);

const dump = spawn(
  "pg_dump",
  ["--format=custom", "--no-owner", "--no-privileges", "--file", outFile, DATABASE_URL],
  { stdio: ["ignore", "inherit", "inherit"] },
);

dump.on("error", (err) => {
  console.error(`[db:backup] pg_dump konnte nicht gestartet werden: ${err.message}`);
  console.error(`[db:backup] Prüfe: which pg_dump`);
  process.exit(1);
});

dump.on("close", (code) => {
  if (code !== 0) {
    console.error(`[db:backup] pg_dump fehlgeschlagen (exit ${code})`);
    process.exit(code ?? 1);
  }
  const sizeMb = (statSync(outFile).size / 1024 / 1024).toFixed(2);
  console.log(`[db:backup] OK – ${outFile} (${sizeMb} MB)`);

  // Retention: nur die neuesten RETENTION .dump-Dateien behalten
  const dumps = readdirSync(BACKUP_DIR)
    .filter((f) => f.endsWith(".dump"))
    .map((f) => ({ f, t: statSync(join(BACKUP_DIR, f)).mtimeMs }))
    .sort((a, b) => b.t - a.t);
  for (const old of dumps.slice(RETENTION)) {
    unlinkSync(join(BACKUP_DIR, old.f));
    console.log(`[db:backup] Alte Sicherung entfernt: ${old.f}`);
  }
});
```
