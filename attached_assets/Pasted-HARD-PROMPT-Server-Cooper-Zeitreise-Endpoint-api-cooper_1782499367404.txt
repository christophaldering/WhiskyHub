  app.post("/api/cooper/timetravel", async (req: Request, res: Response) => {
    const auth = await requireAuth(req);
    if (!auth.authenticated) return res.status(auth.status).json({ message: auth.message });
    const rate = checkConverseRateLimit("timetravel:" + auth.participant.id);
    if (!rate.allowed) return res.status(429).json({ message: "Too many requests.", retryAfter: rate.retryAfterSeconds });
    try {
      const { from, to, label, drams, regenerate } = req.body || {};
      if (!from || !to) return res.status(400).json({ message: "from/to required" });
      const list = Array.isArray(drams) ? drams.slice(0, 200) : [];
      if (list.length === 0) return res.status(400).json({ message: "Keine Drams im Zeitraum." });

      const cacheKey = `timetravel:${auth.participant.id}:${String(from).slice(0, 10)}:${String(to).slice(0, 10)}`;
      if (!regenerate) {
        try {
          const cached = await storage.getAppSetting(cacheKey);
          if (cached) { const c = JSON.parse(cached); return res.json({ ...c, cached: true }); }
        } catch {}
      }

      const isDe = String(req.headers["accept-language"] || "de").toLowerCase().startsWith("de");
      const sorted = [...list].sort((a: any, b: any) => {
        const ta = a?.date ? new Date(a.date).getTime() : 0;
        const tb = b?.date ? new Date(b.date).getTime() : 0;
        return ta - tb;
      });
      const lines = sorted.map((d: any) => {
        const ym = d?.date ? new Date(d.date).toISOString().slice(0, 7) : "????-??";
        const name = String(d?.name || "?").slice(0, 80);
        const dist = d?.distillery ? ` (${String(d.distillery).slice(0, 60)})` : "";
        const meta = [d?.region, d?.cask].filter(Boolean).map((x: any) => String(x).slice(0, 40)).join(", ");
        const sc = `N${d?.nose ?? "-"}/G${d?.taste ?? "-"}/A${d?.finish ?? "-"}/Ø${d?.overall ?? "-"}`;
        const tags = Array.isArray(d?.tags) ? d.tags.slice(0, 10).join(", ") : "";
        const narr = d?.narrative ? String(d.narrative).slice(0, 300) : "";
        return `${ym} | ${name}${dist}${meta ? ` | ${meta}` : ""} | ${sc}${tags ? ` | Tags: ${tags}` : ""}${narr ? ` | ${narr}` : ""}`;
      }).join("\n");

      const periodLabel = label ? String(label).slice(0, 60) : `${String(from).slice(0, 10)} – ${String(to).slice(0, 10)}`;
      const system = `Du bist Cooper, der Küfer — hier im Nachgang der ruhige Kenner. Du erzählst die Verkostungs-REISE EINER Person durch einen Zeitraum: ein zusammenhängender, erzählender Rückblick (KEINE Aufzählung, KEINE Tabelle). Stütze dich AUSSCHLIESSLICH auf die gelieferten Verkostungen; erfinde nichts, ergänze keine Aromen, vergib KEINE eigenen Noten oder Punkte. Beobachte den Bogen: Verschiebungen in Regionen/Fasstypen über die Zeit, wie sich Wortschatz und Wahrnehmung wandeln, auffällige Momente (der am stärksten erlebte Dram, ein Whisky zu dem zurückgekehrt wurde, ein roter Faden). Schreibe in der zweiten Person, ${isDe ? "Deutsch" : "Englisch"}, 200–300 Wörter, warm und beobachtend, nicht schmeichelnd, nicht wertend. Schließe mit EINER neugierigen Frage, die zurück ans Glas einlädt.`;
      const userContent = `ZEITRAUM: ${periodLabel}\nANZAHL VERKOSTUNGEN: ${sorted.length}\n\nVERKOSTUNGEN (älteste zuerst):\n${lines}`;

      const openai = new OpenAI({ apiKey: process.env.AI_INTEGRATIONS_OPENAI_API_KEY, baseURL: process.env.AI_INTEGRATIONS_OPENAI_BASE_URL });
      const completion = await openai.chat.completions.create({
        model: "gpt-5-mini",
        max_completion_tokens: 1100,
        reasoning_effort: "minimal",
        messages: [
          { role: "system", content: system },
          { role: "user", content: userContent },
        ],
      } as any);

      const text = (completion.choices?.[0]?.message?.content || "").trim();
      if (!text) return res.status(502).json({ message: "Leere Antwort vom Modell." });

      const result = { narrative: text, generatedAt: new Date().toISOString(), count: sorted.length, period: periodLabel };
      try { await storage.setAppSetting(cacheKey, JSON.stringify(result)); } catch {}
      return res.json({ ...result, cached: false });
    } catch (e: any) {
      return res.status(500).json({ message: e?.message || "Fehler beim Erzeugen." });
    }
  });
