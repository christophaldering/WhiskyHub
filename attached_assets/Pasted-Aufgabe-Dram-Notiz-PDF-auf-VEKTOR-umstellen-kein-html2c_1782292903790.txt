=== SCHRITT A: client/src/labs/utils/dramNotePdf.ts — gesamten Dateiinhalt durch exakt Folgendes ersetzen: ===
import jsPDF from "jspdf";

export type DramNoteData = {
  whiskyName: string;
  dateISO?: string | null;
  tastingName?: string | null;
  tasterName?: string | null;
  narrative: string;
  scores?: { nose?: number|null; palate?: number|null; finish?: number|null; overall?: number|null };
  photoDataUrl?: string | null;
};

const GOLD: [number,number,number] = [212,168,71];
const AMBER: [number,number,number] = [196,122,58];
const DARK: [number,number,number] = [26,21,16];
const GRAY: [number,number,number] = [133,120,98];
const RULEC: [number,number,number] = [227,217,198];
const SECTIONS = ["Nase","Gaumen","Abgang","Gesamteindruck"];
const EN: Record<string,string> = { Nase:"NOSE", Gaumen:"PALATE", Abgang:"FINISH", Gesamteindruck:"OVERALL" };

function parseSections(text: string){
  const lines=(text||"").split(/\r?\n/); const out:{header:string|null;body:string}[]=[]; let cur:{header:string|null;body:string}|null=null;
  for(const raw of lines){ const line=raw.trim();
    if(SECTIONS.includes(line)){ if(cur)out.push(cur); cur={header:line,body:""}; }
    else { if(!cur)cur={header:null,body:""}; cur.body+=(cur.body?"\n":"")+raw; } }
  if(cur)out.push(cur); return out;
}
function fmtDateTime(iso?: string|null): string {
  if(!iso)return ""; try{ const d=new Date(iso);
    return d.toLocaleDateString("de-DE",{day:"numeric",month:"long",year:"numeric",timeZone:"Europe/Berlin"})+" \u00b7 "+
           d.toLocaleTimeString("de-DE",{hour:"2-digit",minute:"2-digit",timeZone:"Europe/Berlin"})+" Uhr"; }catch{return "";}
}

function buildDramNotePdfDoc(data: DramNoteData): jsPDF {
  const doc = new jsPDF({ unit:"mm", format:"a4" });
  const PW = doc.internal.pageSize.getWidth();
  const PH = doc.internal.pageSize.getHeight();
  const M = 18;
  const CW = PW - M*2;
  const BOTTOM = PH - 16;
  let y = 0;
  const bg = () => { doc.setFillColor(247,242,232); doc.rect(0,0,PW,PH,"F"); };
  bg();
  const ensure = (need: number) => { if (y + need > BOTTOM) { doc.addPage(); bg(); y = M; } };

  y = 20;
  doc.setDrawColor(...GOLD); doc.setLineWidth(0.6); doc.line(M, y, PW-M, y);
  y += 11;
  doc.setFont("times","bold"); doc.setFontSize(30); doc.setTextColor(...GOLD);
  doc.text("CaskSense", PW/2, y, { align:"center" });
  y += 6.5;
  doc.setFont("helvetica","normal"); doc.setFontSize(8); doc.setTextColor(...AMBER);
  doc.text("W H E R E   T A S T I N G   B E C O M E S   R E F L E C T I O N", PW/2, y, { align:"center" });
  y += 5;
  doc.setDrawColor(...GOLD); doc.setLineWidth(0.6); doc.line(M, y, PW-M, y);
  y += 12;

  doc.setFont("helvetica","bold"); doc.setFontSize(8.5); doc.setTextColor(...AMBER);
  doc.text("V E R K O S T U N G S N O T I Z", M, y);
  y += 8;
  doc.setFont("times","bold"); doc.setFontSize(24); doc.setTextColor(...DARK);
  const titleLines = doc.splitTextToSize(data.whiskyName || "Verkostungsnotiz", CW);
  doc.text(titleLines, M, y); y += titleLines.length * 9.5;

  const dt = fmtDateTime(data.dateISO);
  if (dt){ y += 1; doc.setFont("helvetica","normal"); doc.setFontSize(9.5); doc.setTextColor(...GRAY); doc.text(dt, M, y); y += 5.5; }
  if (data.tastingName){ doc.setFont("times","italic"); doc.setFontSize(12); doc.setTextColor(...AMBER); doc.text("Im Tasting \u00ab"+data.tastingName+"\u00bb", M, y); y += 6; }
  if (data.tasterName){ doc.setFont("helvetica","normal"); doc.setFontSize(9.5); doc.setTextColor(...GRAY); doc.text("Verkostet von "+data.tasterName, M, y); y += 6; }

  const sc = data.scores || {};
  const items: Array<[string, number]> = [];
  if (sc.nose != null) items.push(["Nase", sc.nose]);
  if (sc.palate != null) items.push(["Gaumen", sc.palate]);
  if (sc.finish != null) items.push(["Abgang", sc.finish]);
  if (sc.overall != null) items.push(["Gesamt", sc.overall]);
  if (items.length){
    y += 2; let x = M; const baseline = y;
    items.forEach((it,idx)=>{
      doc.setFont("helvetica","normal"); doc.setFontSize(10); doc.setTextColor(...GRAY);
      doc.text(it[0]+" ", x, baseline); x += doc.getTextWidth(it[0]+" ");
      doc.setFont("times","bold"); doc.setFontSize(13); doc.setTextColor(...AMBER);
      doc.text(String(it[1]), x, baseline); x += doc.getTextWidth(String(it[1]));
      if (idx < items.length-1){ doc.setFont("helvetica","normal"); doc.setFontSize(10); doc.setTextColor(...RULEC); doc.text("   \u00b7   ", x, baseline); x += doc.getTextWidth("   \u00b7   "); }
    });
    y = baseline + 6;
  }

  doc.setDrawColor(...RULEC); doc.setLineWidth(0.3); doc.line(M, y, PW-M, y); y += 8;

  if (data.photoDataUrl){
    try {
      const props = doc.getImageProperties(data.photoDataUrl);
      const ph = Math.min(70, (props.height/props.width)*CW);
      ensure(ph + 6);
      doc.addImage(data.photoDataUrl, "JPEG", M, y, CW, ph);
      y += ph + 8;
    } catch(e){ /* Foto ueberspringen */ }
  }

  const blocks = parseSections(data.narrative);
  for (const b of blocks){
    if (b.header){
      ensure(12);
      doc.setFont("times","bold"); doc.setFontSize(14); doc.setTextColor(...GOLD);
      doc.text(b.header, M, y);
      const hw = doc.getTextWidth(b.header);
      doc.setFont("helvetica","bold"); doc.setFontSize(7.5); doc.setTextColor(...GRAY);
      doc.text(EN[b.header]||"", M + hw + 3, y);
      y += 6;
    }
    const body = (b.body||"").trim();
    if (body){
      doc.setFont("times","normal"); doc.setFontSize(11.5); doc.setTextColor(...DARK);
      const lines = doc.splitTextToSize(body, CW);
      for (const ln of lines){ ensure(6); doc.text(ln, M, y); y += 5.4; }
      y += 4;
    }
  }

  const pages = doc.getNumberOfPages();
  for (let p=1; p<=pages; p++){
    doc.setPage(p);
    doc.setDrawColor(...GOLD); doc.setLineWidth(0.5); doc.line(M, PH-14, PW-M, PH-14);
    doc.setFont("helvetica","normal"); doc.setFontSize(7.5); doc.setTextColor(...GRAY);
    doc.text("Where Tasting becomes Reflection  \u00b7  casksense.com", PW/2, PH-9, { align:"center" });
  }
  return doc;
}

export function openDramNotePdf(data: DramNoteData): void {
  const doc = buildDramNotePdfDoc(data);
  const raw = doc.output("blob");
  const blob = raw.type === "application/pdf" ? raw : new Blob([raw], { type: "application/pdf" });
  const url = URL.createObjectURL(blob);
  const win = window.open(url, "_blank");
  if (!win) {
    const a = document.createElement("a");
    a.href = url; a.target = "_blank"; a.rel = "noopener";
    document.body.appendChild(a); a.click(); document.body.removeChild(a);
  }
  setTimeout(() => { URL.revokeObjectURL(url); }, 60000);
}
